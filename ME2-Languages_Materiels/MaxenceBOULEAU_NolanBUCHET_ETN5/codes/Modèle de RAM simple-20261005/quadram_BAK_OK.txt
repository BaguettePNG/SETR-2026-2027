library IEEE;
use IEEE.Std_logic_1164.all;
use IEEE.Numeric_std.all;

entity quadram is
	generic(ADDR_SELECT : INTEGER := 2;
		ADDR_LENGHT : INTEGER := 10;
		DATA_LENGHT : INTEGER := 8
	);
	
    port (
		clock   : in  std_logic;
		we      : in  std_logic;
		address : in  unsigned((ADDR_LENGHT - 1) downto 0);
		datain  : in  unsigned((DATA_LENGHT - 1) downto 0);
		dataout : out unsigned((DATA_LENGHT - 1) downto 0)
    );
end entity quadram;

architecture RTL of quadram is

   	signal CS : unsigned(((2**ADDR_SELECT) - 1) downto 0);
	signal cs_0, cs_1, cs_2, cs_3 : std_logic;
begin

cs_0 <= '1' when CS = "0001" else '0';
cs_1 <= '1' when CS = "0010" else '0';
cs_2 <= '1' when CS = "0100" else '0';
cs_3 <= '1' when CS = "1000" else '0';

DECODER : entity work.decoder(RTL)
generic map (
	SEL_LENGHT => ADDR_SELECT  
)
port map (
	cs      => CS,	-- OUTPUT
	address => address((ADDR_LENGHT - 1) downto ((ADDR_LENGHT - 1) - ( ADDR_SELECT - 1)))	-- INPUT
);

RAM_0 : entity work.sync_ram(RTL)
generic map (
	DataLenght => DATA_LENGHT,
	AddrLenght => ADDR_LENGHT - ADDR_SELECT
)
port map (
	cs			=> cs_0,
    clock   	=> clock,
    we    		=> we,
    address 	=> address(((ADDR_LENGHT - 1) - ADDR_SELECT) downto 0),
    datain  	=> datain,
    dataout     => dataout
);

RAM_1 : entity work.sync_ram(RTL)
generic map (
	DataLenght => DATA_LENGHT,
	AddrLenght => ADDR_LENGHT - ADDR_SELECT
)
port map (
	cs			=> cs_1,
    clock   	=> clock,
    we    		=> we,
    address 	=> address(((ADDR_LENGHT - 1) - ADDR_SELECT) downto 0),
    datain  	=> datain,
    dataout     => dataout
);

RAM_2 : entity work.sync_ram(RTL)
generic map (
	DataLenght => DATA_LENGHT,
	AddrLenght => ADDR_LENGHT - ADDR_SELECT
)
port map (
	cs			=> cs_2,
    clock   	=> clock,
    we    		=> we,
    address 	=> address(((ADDR_LENGHT - 1) - ADDR_SELECT) downto 0),
    datain  	=> datain,
    dataout     => dataout
);

RAM_3 : entity work.sync_ram(RTL)
generic map (
	DataLenght => DATA_LENGHT,
	AddrLenght => ADDR_LENGHT - ADDR_SELECT
)
port map (
	cs			=> cs_3,
    clock   	=> clock,
    we    		=> we,
    address 	=> address(((ADDR_LENGHT - 1) - ADDR_SELECT) downto 0),
    datain  	=> datain,
    dataout     => dataout
);

end architecture RTL;
